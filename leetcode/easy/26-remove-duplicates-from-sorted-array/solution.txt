class Solution:
    def removeDuplicates(self, nums: list[int]) -> int:

        n = len(nums)
        
        unique_position = 1
 
        for current in range(1, n):
            # A different value means a new
            # unique element has been found.
            if nums[current] != nums[unique_position - 1]:
                nums[unique_position] = nums[current]
                unique_position += 1
 
        return unique_position
