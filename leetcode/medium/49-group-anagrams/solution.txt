class Solution(object):
    def groupAnagrams(self, strs):
        """
        :type strs: List[str]
        :rtype: List[List[str]]
        """
        anagram_groups = defaultdict(list)
        for s in strs:
            # Create a sorted character signature as the key
            key = ''.join(sorted(s))
            anagram_groups[key].append(s)
        return list(anagram_groups.values())
