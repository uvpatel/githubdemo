class Solution(object):
    def reverse(self, x):
        is_negative = x < 0

        s = str(abs(x))
        s = s[::-1]

        reversed_num = int(s)

        if is_negative:
            reversed_num = -reversed_num

        ## don't mis
        if reversed_num < -2**31 or reversed_num > 2**31 - 1:
            return 0

        return reversed_num
