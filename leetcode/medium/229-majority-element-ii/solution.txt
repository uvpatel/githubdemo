class Solution:
    def majorityElement(self, nums: list[int]) -> list[int]:
        candidate1 = nums[0]
        candidate2 = nums[0]
        count1 = 0
        count2 = 0

        for value in nums:
            # A match strengthens the first candidate.
            if value == candidate1:
                count1 += 1
